1. Intuition & Architectural Overview

$$ q_m = q e^{im\theta} $$ When the model calculates the dot product between Query $m$ and Key $n$, the math naturally simplifies to only care about the *relative distance* between them ($m - n$).

Processes input tensors through parameterized matrix transformations and non-linear activations.

2. Core Architectural Components

  • Layer Transformations: Parameterized affine mapping and dimensional reshaping
  • Non-Linear Activation: Enforces non-linear feature separation across manifold
  • Residual Optimization: Direct gradient flow preserving signal fidelity

3. Advantages & Trade-Offs

High representational capacity and parameter efficiency, balanced against inductive bias constraints and hardware memory requirements.

class TransformerBlock(nn.Module):
    """
    A single block/layer of the Transformer.
    Modern LLMs stack tens or hundreds of these.
    """
    class="code-kw">def __init__(self, embed_size, heads, dropout, forward_expansion):
        super(TransformerBlock, self).__init__()
        
        self.attention = SelfAttention(embed_size, heads)
        
        class="code-cmt"># Layer Normalization normalizes the outputs of the neurons, keeping variance stable
        self.norm1 = nn.LayerNorm(embed_size)
        self.norm2 = nn.LayerNorm(embed_size)
        
        class="code-cmt"># The Feed-Forward Network (MLP) part of the block
        self.feed_forward = nn.Sequential(
            nn.Linear(embed_size, forward_expansion * embed_size),
            nn.ReLU(),
            nn.Linear(forward_expansion * embed_size, embed_size)
        )
        self.dropout = nn.Dropout(dropout)

    class="code-kw">def forward(self, value, key, query, mask):
        class="code-cmt"># 1. Attention Step
        attention = self.attention(value, key, query, mask)
        
        class="code-cmt"># 2. Add & Norm (Residual connection + LayerNorm)
        class="code-cmt"># We add the original query (input) to the attention output
        x = self.dropout(self.norm1(attention + query))
        
        class="code-cmt"># 3. Feed Forward Step
        forward = self.feed_forward(x)
        
        class="code-cmt"># 4. Add & Norm (Residual connection + LayerNorm)
        out = self.dropout(self.norm2(forward + x))
        
        return out


class TransformerEncoder(nn.Module):
    """
    The full Encoder model (like BERT).
    """
    class="code-kw">def __init__(
        self,
        src_vocab_size,
        embed_size,
        num_layers,
        heads,
        device,
        forward_expansion,
        dropout,
        max_length,
    ):
        super(TransformerEncoder, self).__init__()
        self.embed_size = embed_size
        self.device = device
        
        class="code-cmt"># 1. Word Embedding: Converts token IDs to dense vectors
        self.word_embedding = nn.Embedding(src_vocab_size, embed_size)
        
        class="code-cmt"># 2. Positional Embedding: Transformers have no sense of order. 
        class="code-cmt"># We MUST inject position information mathematically.
        self.position_embedding = nn.Embedding(max_length, embed_size)

        class="code-cmt"># Stack multiple Transformer Blocks
        self.layers = nn.ModuleList(
            [
                TransformerBlock(
                    embed_size,
                    heads,
                    dropout=dropout,
                    forward_expansion=forward_expansion,
                )
                for _ in range(num_layers)
            ]
        )
        self.dropout = nn.Dropout(dropout)

    class="code-kw">def forward(self, x, mask):
        class="code-cmt"># x is a batch of token IDs. Shape: (batch_size, seq_length)
        batch_size, seq_length = x.shape
        
        class="code-cmt"># Create position indices (0, 1, 2, ..., seq_length-1)
        positions = torch.arange(0, seq_length).expand(batch_size, seq_length).to(self.device)
        
        class="code-cmt"># Combine word and positional embeddings
        out = self.dropout(self.word_embedding(x) + self.position_embedding(positions))
        
        class="code-cmt"># Pass through all Transformer blocks
        for layer in self.layers:
            class="code-cmt"># In the encoder, query, key, and value are all the same 'out'
            out = layer(out, out, out, mask)
            
        return out

class GenerativeTransformerModel(nn.Module):
    """
    A full model combining the architecture into something that can predict the next token.
    """
    class="code-kw">def __init__(
        self,
        vocab_size,
        embed_size=256,
        num_layers=6,
        heads=8,
        device="cpu",
        forward_expansion=4,
        dropout=0.1,
        max_length=1000,
    ):
        super(GenerativeTransformerModel, self).__init__()
        
        class="code-cmt"># The Core Processing Engine
        self.encoder = TransformerEncoder(
            vocab_size, embed_size, num_layers, heads, device, forward_expansion, dropout, max_length
        )
        
        class="code-cmt"># The Output Head (maps the final embeddings back to vocab probabilities)
        self.fc_out = nn.Linear(embed_size, vocab_size)
        self.device = device

    class="code-kw">def make_src_mask(self, src):
        class="code-cmt"># Creates a mask to ignore padding tokens (usually token 0)
        src_mask = (src != 0).unsqueeze(1).unsqueeze(2)
        class="code-cmt"># Shape: (N, 1, 1, src_len)
        return src_mask.to(self.device)

    class="code-kw">def forward(self, x):
        src_mask = self.make_src_mask(x)
        class="code-cmt"># Get deep contextual embeddings
        enc_out = self.encoder(x, src_mask)
        class="code-cmt"># Predict next word probabilities
        out = self.fc_out(enc_out)
        return out
